📝 Template Syntax & Directives

Vue.js templates are HTML-based with powerful directives that bind DOM elements to component data. The template syntax is intuitive yet powerful, enabling reactive UI updates with minimal code.

Text Interpolation

<template> <div> <h1>{{ message }}</h1> <p>{{ user.name }} is {{ user.age }} years old</p> <!-- Raw HTML --> <div v-html="rawHtml"></div> <!-- Attribute binding --> <img :src="imageUrl" :alt="imageAlt"> <button :disabled="isDisabled">Click me</button> <!-- JavaScript expressions --> <p>{{ message.split('').reverse().join('') }}</p> <p>{{ isSeen ? 'Yes' : 'No' }}</p> </div> </template> <script setup> import { ref } from 'vue' const message = ref('Hello Vue!') const user = ref({ name: 'John', age: 25 }) const rawHtml = ref('<strong>Bold text</strong>') const imageUrl = ref('/path/to/image.jpg') const imageAlt = ref('Description') const isDisabled = ref(false) const isSeen = ref(true) </script>

Directives

<template> <!-- Conditional rendering --> <p v-if="seen">Now you see me</p> <p v-else-if="type === 'A'">A</p> <p v-else>Neither</p> <!-- Show/hide (CSS display) --> <p v-show="isVisible">Toggle visibility</p> <!-- List rendering --> <ul> <li v-for="(item, index) in items" :key="item.id"> {{ index }}: {{ item.text }} </li> </ul> <!-- Event handling --> <button @click="doSomething">Click me</button> <button @click="doThat('hello', $event)">With arguments</button> <!-- Modifiers --> <form @submit.prevent="onSubmit"> <input @keyup.enter="submit"> </form> <!-- Two-way binding --> <input v-model="message"> <textarea v-model="message"></textarea> </template>

🎯 Composition API Deep Dive

The Composition API is the new standard in Vue.js 3, providing a more flexible and powerful way to organize component logic. It enables better code reuse, TypeScript support, and logical organization.

Basic Composition API

<script setup> import { ref, computed, watch, onMounted } from 'vue' // Reactive state const count = ref(0) const message = ref('Hello Vue 3!') // Computed properties const doubledCount = computed(() => count.value * 2) const greeting = computed(() => `${message.value} Count is ${count.value}`) // Methods function increment() { count.value++ } function decrement() { count.value-- } // Watchers watch(count, (newValue, oldValue) => { console.log(`Count changed from ${oldValue} to ${newValue}`) }) // Lifecycle hooks onMounted(() => { console.log('Component mounted!') }) // Expose to template defineExpose({ count, message }) </script>

Composables (Reusable Logic)

// composables/useCounter.js import { ref, computed } from 'vue' export function useCounter(initialValue = 0) { const count = ref(initialValue) const doubled = computed(() => count.value * 2) function increment() { count.value++ } function decrement() { count.value-- } function reset() { count.value = initialValue } return { count, doubled, increment, decrement, reset } } // Using the composable <script setup> import { useCounter } from './composables/useCounter' const { count, doubled, increment, decrement, reset } = useCounter(10) </script>

Advanced Composition Patterns

// composables/useFetch.js import { ref, watchEffect } from 'vue' export function useFetch(url) { const data = ref(null) const error = ref(null) const loading = ref(true) async function fetchData() { loading.value = true error.value = null try { const response = await fetch(url.value) data.value = await response.json() } catch (err) { error.value = err } finally { loading.value = false } } // Refetch when URL changes watchEffect(fetchData) return { data, error, loading, refetch: fetchData } } // Usage <script setup> import { useFetch } from './composables/useFetch' const searchQuery = ref('vue') const url = computed(() => `/api/search?q=${searchQuery.value}`) const { data, error, loading } = useFetch(url) </script>

🔄 Reactivity System

Vue.js 3's reactivity system is built on Proxies, providing better performance and more capabilities than the Object.defineProperty-based system in Vue.js 2.

Reactive References

<script setup> import { ref, reactive, readonly, toRefs, toRef } from 'vue' // ref() for primitive values const count = ref(0) const message = ref('Hello') // reactive() for objects const user = reactive({ name: 'John', age: 25, address: { city: 'New York', country: 'USA' } }) // readonly() to prevent modifications const readonlyUser = readonly(user) // toRefs() to destructure reactive objects const { name, age } = toRefs(user) // toRef() for single property const city = toRef(user.address, 'city') // Shallow ref for performance optimization const largeData = shallowRef({ /* large object */ }) // Custom ref with getter/setter function useDebouncedRef(value, delay = 200) { let timeout return customRef((track, trigger) => ({ get() { track() return value }, set(newValue) { clearTimeout(timeout) timeout = setTimeout(() => { value = newValue trigger() }, delay) } })) } const debouncedText = useDebouncedRef('hello') </script>

Computed and Watch

<script setup> import { ref, computed, watch, watchEffect } from 'vue' const firstName = ref('John') const lastName = ref('Doe') const age = ref(25) // Computed properties const fullName = computed({ get: () => `${firstName.value} ${lastName.value}`, set: (value) => { [firstName.value, lastName.value] = value.split(' ') } }) const isAdult = computed(() => age.value >= 18) // Watch for specific changes watch(firstName, (newName, oldName) => { console.log(`First name changed from ${oldName} to ${newName}`) }) // Watch multiple sources watch([firstName, lastName], ([newFirst, newLast], [oldFirst, oldLast]) => { console.log('Name changed:', `${oldFirst} ${oldLast}` → `${newFirst} ${newLast}`) }) // WatchEffect for immediate execution watchEffect(() => { console.log(`Current full name: ${fullName.value}`) }) // Watch with options watch(age, (newAge) => { if (newAge >= 18) { console.log('User is now an adult!') } }, { immediate: true }) // Watch for deep changes const user = reactive({ profile: { name: 'John' } }) watch(() => user.profile.name, (newName) => { console.log('Profile name changed:', newName) }, { deep: true }) </script>

🧩 Components & Props

Components are the building blocks of Vue.js applications. They encapsulate HTML, CSS, and JavaScript into reusable pieces, promoting code organization and maintainability.

Creating Components

<template> <button :class="buttonClass" :disabled="disabled" @click="$emit('click', $event)" > <slot>{{ text }}</slot> </button> </template> <script setup> import { computed } from 'vue' const props = defineProps({ text: { type: String, default: 'Button' }, variant: { type: String, default: 'primary', validator: (value) => ['primary', 'secondary', 'danger'].includes(value) }, disabled: { type: Boolean, default: false } }) const emit = defineEmits(['click']) const buttonClass = computed(() => [ 'btn', `btn-${props.variant}`, { 'btn-disabled': props.disabled } ]) </script> <style scoped> .btn { padding: 8px 16px; border: none; border-radius: 4px; cursor: pointer; } .btn-primary { background-color: #42b883; color: white; } .btn-secondary { background-color: #6c757d; color: white; } .btn-danger { background-color: #dc3545; color: white; } .btn-disabled { opacity: 0.6; cursor: not-allowed; } </style>

Component Communication

<template> <div> <ChildComponent :message="parentMessage" :user="currentUser" @update-message="handleUpdate" @custom-event="handleCustomEvent" /> <!-- Provide/Inject for deep nesting --> <DeepChild /> </div> </template> <script setup> import { provide, ref } from 'vue' import ChildComponent from './ChildComponent.vue' import DeepChild from './DeepChild.vue' const parentMessage = ref('Hello from parent') const currentUser = ref({ name: 'John', id: 1 }) // Provide for descendant components provide('theme', 'dark') provide('user', currentUser) function handleUpdate(newMessage) { parentMessage.value = newMessage } function handleCustomEvent(payload) { console.log('Custom event received:', payload) } </script> <script setup> import { inject } from 'vue' const props = defineProps(['message', 'user']) const emit = defineEmits(['update-message', 'custom-event']) const theme = inject('theme') const user = inject('user') function sendMessage() { emit('update-message', 'Message from child') emit('custom-event', { type: 'custom', data: 'hello' }) } </script>

Dynamic Components

<template> <div> <component :is="currentComponent" :key="componentKey" v-bind="currentProps" /> <!-- Keep alive for component state --> <KeepAlive> <component :is="currentComponent" /> </KeepAlive> </div> </template> <script setup> import { ref, shallowRef } from 'vue' import ComponentA from './ComponentA.vue' import ComponentB from './ComponentB.vue' const currentComponent = shallowRef(ComponentA) const componentKey = ref(0) const currentProps = ref({}) function switchToComponentB() { currentComponent.value = ComponentB currentProps.value = { message: 'Hello B' } componentKey.value++ } </script>

🗃️ Pinia State Management

Pinia is the official state management library for Vue.js, providing a simple, intuitive, and type-safe solution for managing application state.

Setting Up Pinia

// stores/index.js import { createPinia } from 'pinia' const pinia = createPinia() export default pinia // main.js import { createApp } from 'vue' import App from './App.vue' import pinia from './stores' const app = createApp(App) app.use(pinia) app.mount('#app')

Creating Stores

// stores/counter.js import { defineStore } from 'pinia' export const useCounterStore = defineStore('counter', { state: () => ({ count: 0, name: 'Eduardo' }), getters: { doubleCount: (state) => state.count * 2, doubleCountPlusOne: () => this.doubleCount + 1, // Pass arguments to getters getUserById: (state) => { return (userId) => state.users.find((user) => user.id === userId) } }, actions: { increment() { this.count++ }, async fetchUserData() { try { const response = await fetch('/api/user') const userData = await response.json() this.setUser(userData) } catch (error) { console.error('Error fetching user:', error) } }, setUser(userData) { this.name = userData.name this.count = userData.count } } }) // Using the store <script setup> import { useCounterStore } from '@/stores/counter' const counterStore = useCounterStore() // Access state console.log(counterStore.count) // Access getters console.log(counterStore.doubleCount) // Call actions counterStore.increment() counterStore.fetchUserData() // Destructure with reactivity const { count, doubleCount } = storeToRefs(counterStore) </script>

Advanced Pinia Features

// stores/cart.js import { defineStore } from 'pinia' export const useCartStore = defineStore('cart', { state: () => ({ items: [], checkoutStatus: null }), getters: { cartTotal: (state) => { return state.items.reduce((total, item) => { return total + item.price * item.quantity }, 0) }, itemCount: (state) => { return state.items.reduce((count, item) => count + item.quantity, 0) } }, actions: { addToCart(product) { const existingItem = this.items.find(item => item.id === product.id) if (existingItem) { existingItem.quantity++ } else { this.items.push({ ...product, quantity: 1 }) } }, removeFromCart(productId) { const index = this.items.findIndex(item => item.id === productId) if (index > -1) { this.items.splice(index, 1) } }, updateQuantity(productId, quantity) { const item = this.items.find(item => item.id === productId) if (item) { item.quantity = Math.max(0, quantity) } }, async checkout() { this.checkoutStatus = 'loading' try { const response = await fetch('/api/checkout', { method: 'POST', body: JSON.stringify({ items: this.items }) }) if (response.ok) { this.checkoutStatus = 'success' this.items = [] } else { this.checkoutStatus = 'failed' } } catch (error) { this.checkoutStatus = 'failed' console.error('Checkout failed:', error) } } } })

🛣️ Vue Router

Vue Router is the official router for Vue.js, enabling single-page applications with navigation, route guards, and nested routes.

Basic Router Setup

// router/index.js import { createRouter, createWebHistory } from 'vue-router' import Home from '../views/Home.vue' import About from '../views/About.vue' import User from '../views/User.vue' const routes = [ { path: '/', name: 'Home', component: Home }, { path: '/about', name: 'About', component: About }, { path: '/user/:id', name: 'User', component: User, props: true }, { path: '/:pathMatch(.*)*', name: 'NotFound', component: () => import('../views/NotFound.vue') } ] const router = createRouter({ history: createWebHistory(), routes }) export default router

Navigation and Route Parameters

<template> <nav> <router-link to="/">Home</router-link> <router-link to="/about">About</router-link> <router-link :to="{ name: 'User', params: { id: 123 }}">User 123</router-link> <!-- Programmatic navigation --> <button @click="goToAbout">Go to About</button> <button @click="goBack">Go Back</button> </nav> <!-- Route outlet --> <router-view /> </template> <script setup> import { useRouter, useRoute } from 'vue-router' const router = useRouter() const route = useRoute() function goToAbout() { router.push('/about') // or router.push({ name: 'About' }) } function goBack() { router.go(-1) } // Access route parameters const userId = computed(() => route.params.id) const query = computed(() => route.query) </script>

Route Guards and Navigation

// router/index.js router.beforeEach((to, from) => { // Check authentication if (to.meta.requiresAuth && !isAuthenticated()) { return { name: 'Login', query: { redirect: to.fullPath } } } // Set page title document.title = to.meta.title || 'Default Title' }) // In component <script setup> import { onBeforeRouteLeave, onBeforeRouteUpdate } from 'vue-router' // Leave guard onBeforeRouteLeave((to, from) => { if (hasUnsavedChanges) { const answer = window.confirm('Do you really want to leave?') if (!answer) return false } }) // Update guard onBeforeRouteUpdate((to, from) => { if (to.params.id !== from.params.id) { fetchData(to.params.id) } }) </script> // Nested routes const routes = [ { path: '/user/:id', component: User, children: [ { path: 'profile', component: UserProfile }, { path: 'posts', component: UserPosts } ] } ]

🚀 Advanced Vue.js Features

Vue.js 3 includes advanced features for building sophisticated applications, including custom directives, plugins, and performance optimization techniques.

Custom Directives

// directives/focus.js const focus = { mounted(el) { el.focus() } } // directives/lazy-load.js const lazyLoad = { mounted(el, binding) { const observer = new IntersectionObserver((entries) => { entries.forEach(entry => { if (entry.isIntersecting) { el.src = binding.value observer.unobserve(el) } }) }) observer.observe(el) } } // Register globally app.directive('focus', focus) app.directive('lazy-load', lazyLoad) // Usage <template> <input v-focus /> <img v-lazy-load="imageUrl" /> </template>

Plugins

// plugins/i18n.js export default { install: (app, options) => { app.config.globalProperties.$translate = (key) => { return key.split('.').reduce((o, i) => o[i], options.messages) } app.provide('i18n', options) } } // Use plugin app.use(i18n, { messages: { hello: 'Hello World', greeting: { morning: 'Good morning' } } }) // In component <script setup> import { inject } from 'vue' const i18n = inject('i18n') const { $translate } = getCurrentInstance().appContext.config.globalProperties </script>

Performance Optimization

// Lazy loading components const AsyncComponent = defineAsyncComponent(() => import('./components/HeavyComponent.vue') ) // With loading and error states const AsyncComponent = defineAsyncComponent({ loader: () => import('./components/HeavyComponent.vue'), loadingComponent: LoadingComponent, errorComponent: ErrorComponent, delay: 200, timeout: 3000 }) // Teleport for modals <template> <button @click="showModal = true">Show Modal</button> <Teleport to="body"> <div v-if="showModal" class="modal"> <div class="modal-content"> <p>Modal content</p> <button @click="showModal = false">Close</button> </div> </div> </Teleport> </template> // Suspense for async components <template> <Suspense> <template #default> <AsyncComponent /> </template> <template #fallback> <div>Loading...</div> </template> </Suspense> </template>
