Advanced SQL Queries

-- Basic SELECT with complex JOINs SELECT u.user_id, u.username, u.email, p.profile_name, o.order_id, o.order_date, o.total_amount, oi.product_name, oi.quantity, oi.unit_price, c.category_name FROM users u LEFT JOIN user_profiles p ON u.user_id = p.user_id INNER JOIN orders o ON u.user_id = o.user_id INNER JOIN order_items oi ON o.order_id = oi.order_id INNER JOIN products pr ON oi.product_id = pr.product_id INNER JOIN categories c ON pr.category_id = c.category_id WHERE o.order_date >= '2024-01-01' AND o.status = 'completed' ORDER BY o.order_date DESC, o.total_amount DESC; -- Window Functions for Analytics SELECT order_id, user_id, order_date, total_amount, ROW_NUMBER() OVER (ORDER BY total_amount DESC) as rank_by_amount, DENSE_RANK() OVER (PARTITION BY user_id ORDER BY order_date) as order_sequence, LAG(total_amount, 1) OVER (PARTITION BY user_id ORDER BY order_date) as previous_order_amount, SUM(total_amount) OVER (PARTITION BY user_id ORDER BY order_date ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) as running_total, AVG(total_amount) OVER (PARTITION BY user_id ORDER BY order_date ROWS BETWEEN 2 PRECEDING AND 2 FOLLOWING) as moving_avg FROM orders WHERE order_date >= '2024-01-01'; -- Common Table Expressions (CTEs) for Complex Queries WITH monthly_sales AS ( SELECT DATE_TRUNC('month', order_date) as month, COUNT(DISTINCT user_id) as unique_customers, COUNT(order_id) as total_orders, SUM(total_amount) as total_revenue, AVG(total_amount) as avg_order_value FROM orders WHERE order_date >= '2024-01-01' GROUP BY DATE_TRUNC('month', order_date) ), monthly_growth AS ( SELECT month, unique_customers, total_orders, total_revenue, avg_order_value, LAG(total_revenue) OVER (ORDER BY month) as previous_revenue, LAG(unique_customers) OVER (ORDER BY month) as previous_customers FROM monthly_sales ) SELECT month, unique_customers, total_orders, total_revenue, avg_order_value, ROUND((total_revenue - previous_revenue) / previous_revenue * 100, 2) as revenue_growth_pct, ROUND((unique_customers - previous_customers) / previous_customers * 100, 2) as customer_growth_pct FROM monthly_growth ORDER BY month; -- Stored Procedure for Order Processing CREATE PROCEDURE process_order( IN p_user_id INT, IN p_order_items JSON, IN p_shipping_address JSON, OUT p_order_id INT, OUT p_status VARCHAR(50) ) BEGIN DECLARE v_total_amount DECIMAL(10,2) DEFAULT 0; DECLARE v_item_count INT DEFAULT 0; DECLARE v_product_id INT; DECLARE v_quantity INT; DECLARE v_unit_price DECIMAL(10,2); DECLARE v_stock_quantity INT; DECLARE i INT DEFAULT 0; -- Start transaction START TRANSACTION; -- Calculate total amount and validate stock WHILE i < JSON_LENGTH(p_order_items) DO SET v_product_id = JSON_UNQUOTE(JSON_EXTRACT(p_order_items, CONCAT('$[', i, '].product_id'))); SET v_quantity = JSON_UNQUOTE(JSON_EXTRACT(p_order_items, CONCAT('$[', i, '].quantity'))); -- Get product price and stock SELECT unit_price, stock_quantity INTO v_unit_price, v_stock_quantity FROM products WHERE product_id = v_product_id FOR UPDATE; -- Validate stock IF v_stock_quantity < v_quantity THEN SET p_status = CONCAT('Insufficient stock for product ', v_product_id); ROLLBACK; LEAVE main_proc; END IF; -- Update stock UPDATE products SET stock_quantity = stock_quantity - v_quantity WHERE product_id = v_product_id; -- Calculate subtotal SET v_total_amount = v_total_amount + (v_unit_price * v_quantity); SET v_item_count = v_item_count + 1; SET i = i + 1; END WHILE; -- Create order INSERT INTO orders (user_id, order_date, total_amount, status, item_count) VALUES (p_user_id, NOW(), v_total_amount, 'pending', v_item_count); SET p_order_id = LAST_INSERT_ID(); -- Create order items SET i = 0; WHILE i < JSON_LENGTH(p_order_items) DO SET v_product_id = JSON_UNQUOTE(JSON_EXTRACT(p_order_items, CONCAT('$[', i, '].product_id'))); SET v_quantity = JSON_UNQUOTE(JSON_EXTRACT(p_order_items, CONCAT('$[', i, '].quantity'))); SELECT unit_price INTO v_unit_price FROM products WHERE product_id = v_product_id; INSERT INTO order_items (order_id, product_id, quantity, unit_price) VALUES (p_order_id, v_product_id, v_quantity, v_unit_price); SET i = i + 1; END WHILE; -- Create shipping address INSERT INTO shipping_addresses (order_id, address_line1, address_line2, city, state, postal_code, country) VALUES ( p_order_id, JSON_UNQUOTE(JSON_EXTRACT(p_shipping_address, '$.address_line1')), JSON_UNQUOTE(JSON_EXTRACT(p_shipping_address, '$.address_line2')), JSON_UNQUOTE(JSON_EXTRACT(p_shipping_address, '$.city')), JSON_UNQUOTE(JSON_EXTRACT(p_shipping_address, '$.state')), JSON_UNQUOTE(JSON_EXTRACT(p_shipping_address, '$.postal_code')), JSON_UNQUOTE(JSON_EXTRACT(p_shipping_address, '$.country')) ); COMMIT; SET p_status = 'Order processed successfully'; main_proc: BEGIN END; END; -- Trigger for Audit Logging CREATE TABLE order_audit ( audit_id INT AUTO_INCREMENT PRIMARY KEY, order_id INT, old_status VARCHAR(50), new_status VARCHAR(50), changed_by VARCHAR(100), changed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, change_reason TEXT ); CREATE TRIGGER order_status_audit AFTER UPDATE ON orders FOR EACH ROW BEGIN IF OLD.status != NEW.status THEN INSERT INTO order_audit (order_id, old_status, new_status, changed_by, change_reason) VALUES (NEW.order_id, OLD.status, NEW.status, CURRENT_USER(), 'Status updated'); END IF; END; -- Performance Optimization with Indexes -- Create composite index for common query patterns CREATE INDEX idx_orders_user_date_status ON orders(user_id, order_date DESC, status); CREATE INDEX idx_products_category_stock ON products(category_id, stock_quantity); CREATE INDEX idx_order_items_order_product ON order_items(order_id, product_id); -- Partitioning for large tables -- Partition orders table by year ALTER TABLE orders PARTITION BY RANGE (YEAR(order_date)) ( PARTITION p2022 VALUES LESS THAN (2023), PARTITION p2023 VALUES LESS THAN (2024), PARTITION p2024 VALUES LESS THAN (2025), PARTITION p2025 VALUES LESS THAN (2026), PARTITION p_future VALUES LESS THAN MAXVALUE );

  • ✅ Regularly monitor and optimize database performance
  • ✅ Document database schema and design decisions
