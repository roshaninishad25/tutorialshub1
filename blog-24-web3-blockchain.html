// SPDX-License-Identifier: MIT pragma solidity ^0.8.19; // Basic contract structure contract HelloWorld { // State variables string public message = "Hello, World!"; uint public counter = 0; address public owner; // Events event MessageUpdated(string newMessage, address updatedBy); event CounterIncremented(uint newCounter); // Constructor constructor() { owner = msg.sender; } // Basic functions function setMessage(string memory _message) public { require(msg.sender == owner, "Only owner can change message"); message = _message; emit MessageUpdated(_message, msg.sender); } function incrementCounter() public { counter++; emit CounterIncremented(counter); } // View and pure functions function getMessage() public view returns (string memory) { return message; } function add(uint a, uint b) public pure returns (uint) { return a + b; } } // Advanced data types and structures contract AdvancedTypes { // Arrays uint[] public numbers; string[] public names; // Fixed-size arrays uint[5] public fixedNumbers; // Mappings mapping(address => uint) public balances; mapping(string => address) public addressesByName; // Structs struct User { string name; uint age; bool isActive; } mapping(address => User) public users; User[] public userList; // Enums enum Status { Pending, Active, Inactive, Suspended } Status public currentStatus; // Functions with complex types function addUser(string memory _name, uint _age) public { User memory newUser = User({ name: _name, age: _age, isActive: true }); users[msg.sender] = newUser; userList.push(newUser); } function updateBalance(uint _amount) public { balances[msg.sender] += _amount; } function getBalance(address _user) public view returns (uint) { return balances[_user]; } } // Inheritance and modifiers contract AccessControl { address public owner; mapping(address => bool) public authorized; event OwnershipTransferred(address indexed previousOwner, address indexed newOwner); event AuthorizationChanged(address indexed user, bool authorized); modifier onlyOwner() { require(msg.sender == owner, "Only owner can call this function"); _; } modifier onlyAuthorized() { require(authorized[msg.sender] || msg.sender == owner, "Not authorized"); _; } constructor() { owner = msg.sender; authorized[msg.sender] = true; } function transferOwnership(address _newOwner) public onlyOwner { require(_newOwner != address(0), "Invalid address"); emit OwnershipTransferred(owner, _newOwner); owner = _newOwner; } function setAuthorized(address _user, bool _authorized) public onlyOwner { authorized[_user] = _authorized; emit AuthorizationChanged(_user, _authorized); } } contract Bank is AccessControl { mapping(address => uint) public deposits; event Deposited(address indexed user, uint amount); event Withdrawn(address indexed user, uint amount); function deposit() public payable { require(msg.value > 0, "Must send ETH"); deposits[msg.sender] += msg.value; emit Deposited(msg.sender, msg.value); } function withdraw(uint _amount) public onlyAuthorized { require(_amount <= deposits[msg.sender], "Insufficient balance"); deposits[msg.sender] -= _amount; payable(msg.sender).transfer(_amount); emit Withdrawn(msg.sender, _amount); } function getBalance() public view returns (uint) { return deposits[msg.sender]; } }

payable(msg.sender).transfer(_amount); } function transfer(address _to, uint _amount) public { if (_to == address(0)) { revert InvalidAddress(_to); } if (balances[msg.sender] < _amount) { revert InsufficientBalance(_amount, balances[msg.sender]); Best Practices

  • ✅ Always follow the Checks-Effects-Interactions pattern
  • ✅ Use proper access control and authorization mechanisms
  • ✅ Implement comprehensive error handling with custom errors
  • ✅ Optimize gas usage with efficient data structures
  • ✅ Thoroughly test smart contracts before deployment
  • ✅ Use established libraries and standards like OpenZeppelin
  • ✅ Implement proper event logging for off-chain monitoring
  • ✅ Consider upgradeability patterns for long-term contracts
