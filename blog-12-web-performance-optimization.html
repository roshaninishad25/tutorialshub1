📊 Core Web Vitals & Metrics

Core Web Vitals are essential metrics that measure user experience. Understanding and optimizing these metrics is crucial for modern web development.

Largest Contentful Paint (LCP)

🖼️

LCP: Largest Contentful Paint

Measures loading performance. Time from when the page starts loading to when the largest content element appears in the viewport.

Target: Under 2.5 seconds

⏳ Lazy Loading Strategies

Lazy loading defers loading of non-critical resources until they're needed, significantly improving initial page load time and reducing bandwidth usage.

Component Lazy Loading

// React.lazy for component lazy loading import React, { Suspense } from 'react'; const LazyComponent = React.lazy(() => import('./LazyComponent')); function App() { return ( <div> <Suspense fallback={<div>Loading...</div>}> <LazyComponent /> </Suspense> </div> ); } // Vue.js async components const LazyComponent = () => import('./LazyComponent.vue'); new Vue({ components: { LazyComponent } }); // Dynamic imports for JavaScript modules async function loadModule() { const module = await import('./heavy-module.js'); module.doSomething(); } // Intersection Observer for component loading class ComponentLoader { constructor(element, loader) { this.element = element; this.loader = loader; this.observer = new IntersectionObserver(this.handleIntersection.bind(this)); this.observer.observe(element); } async handleIntersection(entries) { if (entries[0].isIntersecting) { const component = await this.loader(); this.element.appendChild(component); this.observer.unobserve(this.element); } } }

// Custom RUM implementation class PerformanceMonitor { constructor(apiEndpoint) { this.apiEndpoint = apiEndpoint; this.metrics = {}; this.init(); } init() { // Monitor Core Web Vitals this.observeLCP(); this.observeFID(); this.observeCLS(); // Monitor custom metrics this.observeResourceTiming(); this.observeNavigationTiming(); // Send data on page unload window.addEventListener('beforeunload', () => { this.sendMetrics(); }); } observeLCP() { new PerformanceObserver((entryList) => { const entries = entryList.getEntries(); const lastEntry = entries[entries.length - 1]; this.metrics.lcp = lastEntry.startTime; }).observe({ entryTypes: ['largest-contentful-paint'] }); } observeFID() { new PerformanceObserver((entryList) => { for (const entry of entryList.getEntries()) { this.metrics.fid = entry.processingStart - entry.startTime; } }).observe({ entryTypes: ['first-input'] }); } observeCLS() { let clsValue = 0; new PerformanceObserver((entryList) => { for (const entry of entryList.getEntries()) { if (!entry.hadRecentInput) { clsValue += entry.value; } } this.metrics.cls = clsValue; }).observe({ entryTypes: ['layout-shift'] }); } observeResourceTiming() { const resources = performance.getEntriesByType('resource'); this.metrics.resources = resources.map(resource => ({ name: resource.name, duration: resource.duration, size: resource.transferSize, type: resource.initiatorType })); } observeNavigationTiming() { const navigation = performance.getEntriesByType('navigation')[0]; this.metrics.navigation = { dns: navigation.domainLookupEnd - navigation.domainLookupStart, tcp: navigation.connectEnd - navigation.connectStart, ssl: navigation.secureConnectionStart > 0 ? navigation.connectEnd - navigation.secureConnectionStart : 0, ttfb: navigation.responseStart - navigation.requestStart, domLoad: navigation.domContentLoadedEventEnd - navigation.domContentLoadedEventStart, windowLoad: navigation.loadEventEnd - navigation.loadEventStart }; } async sendMetrics() { try { await navigator.sendBeacon(this.apiEndpoint, JSON.stringify({ url: window.location.href, userAgent: navigator.userAgent, timestamp: Date.now(), metrics: this.metrics })); } catch (error) { console.error('Failed to send metrics:', error); } } } // Initialize monitoring const monitor = new PerformanceMonitor('/api/performance'); // Google Analytics integration function sendToGoogleAnalytics(metrics) { gtag('event', 'web_vitals', { event_category: 'Web Vitals', event_label: 'LCP', value: Math.round(metrics.lcp), non_interaction: true }); gtag('event', 'web_vitals', { event_category: 'Web Vitals', event_label: 'FID', value: Math.round(metrics.fid), non_interaction: true }); gtag('event', 'web_vitals', { event_category: 'Web Vitals', event_label: 'CLS', value: Math.round(metrics.cls * 1000), non_interaction: true }); }

// Performance alerting system class PerformanceAlerts { constructor(webhookUrl) { this.webhookUrl = webhookUrl; this.thresholds = { lcp: 2500, fid: 100, cls: 0.1, bundleSize: 250 * 1024 }; } checkMetrics(metrics) { const alerts = []; if (metrics.lcp > this.thresholds.lcp) { alerts.push({ type: 'LCP', value: metrics.lcp, threshold: this.thresholds.lcp, severity: 'high' }); } if (metrics.fid > this.thresholds.fid) { alerts.push({ type: 'FID', value: metrics.fid, threshold: this.thresholds.fid, severity: 'medium' }); } if (metrics.cls > this.thresholds.cls) { alerts.push({ type: 'CLS', value: metrics.cls, threshold: this.thresholds.cls, severity: 'high' }); } if (alerts.length > 0) { this.sendAlert(alerts, metrics); } } async sendAlert(alerts, metrics) { const payload = { text: '🚨 Performance Alert', attachments: alerts.map(alert => ({ color: alert.severity === 'high' ? 'danger' : 'warning', fields: [ { title: alert.type, value: `Actual: ${alert.value}ms\nThreshold: ${alert.threshold}ms`, short: true } ] })) }; try { await fetch(this.webhookUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); } catch (error) { console.error('Failed to send alert:', error); } } } // Performance dashboard data aggregation class PerformanceDashboard { constructor() { this.data = []; } addMetrics(metrics) { this.data.push({ timestamp: Date.now(), url: window.location.href, ...metrics }); // Keep only last 1000 entries if (this.data.length > 1000) { this.data = this.data.slice(-1000); } } getAverageMetrics(timeRange = 3600000) { // 1 hour default const cutoff = Date.now() - timeRange; const recentData = this.data.filter(d => d.timestamp > cutoff); return { lcp: this.average(recentData.map(d => d.lcp).filter(Boolean)), fid: this.average(recentData.map(d => d.fid).filter(Boolean)), cls: this.average(recentData.map(d => d.cls).filter(Boolean)) }; } average(arr) { return arr.length > 0 ? arr.reduce((a, b) => a + b, 0) / arr.length : 0; } getPercentile(metric, percentile = 95) { const values = this.data .map(d => d[metric]) .filter(Boolean) .sort((a, b) => a - b); const index = Math.ceil((percentile / 100) * values.length) - 1; return values[index] || 0; } }
