🎯 Advanced Generics

Master TypeScript's powerful generic system to create reusable, type-safe components and utilities. Learn about generic constraints, conditional types, and advanced generic patterns.

Generic Functions and Classes

// Basic generic function function identity<T>(arg: T): T { return arg; } // Generic with constraints interface Lengthwise { length: number; } function logLength<T extends Lengthwise>(arg: T): void { console.log(arg.length); } logLength("hello"); // Works logLength([1, 2, 3]); // Works // logLength(42); // Error: number doesn't have length // Generic class class Box<T> { private content: T; constructor(content: T) { this.content = content; } getContent(): T { return this.content; } setContent(content: T): void { this.content = content; } } const stringBox = new Box("Hello TypeScript"); const numberBox = new Box(42); // Advanced generic with multiple type parameters class Pair<T, U> { constructor( public first: T, public second: U ) {} swap(): Pair<U, T> { return new Pair(this.second, this.first); } toString(): string { return `Pair(${this.first}, ${this.second})`; } } const pair = new Pair("hello", 42); const swapped = pair.swap(); // Pair<number, string> // Generic factory pattern interface Constructor<T> { new(...args: any[]): T; } function createInstance<T>( ctor: Constructor<T>, ...args: any[] ): T { return new ctor(...args); } class User { constructor(public name: string, public age: number) {} } const user = createInstance(User, "John", 30); // Generic utility functions function getProperty<T, K extends keyof T>( obj: T, key: K ): T[K] { return obj[key]; } function setProperty<T, K extends keyof T>( obj: T, key: K, value: T[K] ): void { obj[key] = value; } const person = { name: "Alice", age: 25 }; const name = getProperty(person, "name"); // string setProperty(person, "age", 26);

Conditional Types and Inference

// Conditional types type IsString<T> = T extends string ? true : false; type Test1 = IsString<"hello">; // true type Test2 = IsString<42>; // false // Advanced conditional types type NonNullable<T> = T extends null | undefined ? never : T; type Test3 = NonNullable<string | null>; // string type Test4 = NonNullable<number | undefined>; // number // Conditional type with union distribution type ToArray<T> = T extends any ? T[] : never; type Test5 = ToArray<string | number>; // string[] | number[] // Infer keyword in conditional types type ReturnType<T> = T extends (...args: any[]) => infer R ? R : any; type Test6 = ReturnType<() => string>; // string type Test7 = ReturnType<() => number>; // number // Extract parameter types type FirstParameter<T> = T extends ( first: infer F, ...args: any[] ) => any ? F : never; type Test8 = FirstParameter<(name: string, age: number) => void>; // string // Advanced inference with tuples type UnpackTuple<T> = T extends (infer U)[] ? U : never; type Test9 = UnpackTuple<[string, number, boolean]>; // string | number | boolean // Generic type guards function isString<T>(value: T): value is Extract<T, string> { return typeof value === "string"; } function processValue<T>(value: T): void { if (isString(value)) { // TypeScript knows value is string here console.log(value.toUpperCase()); } } // Generic constraints with keyof function getPropertyValue<T, K extends keyof T>( obj: T, key: K ): T[K] { return obj[key]; } // Generic with default types interface ApiResponse<T = any> { data: T; status: number; message: string; } const response: ApiResponse = { data: { id: 1, name: "Test" }, status: 200, message: "Success" }; // Generic recursive types type DeepPartial<T> = { [P in keyof T]?: T[P] extends object ? DeepPartial<T[P]> : T[P]; }; interface User { id: number; name: string; address: { street: string; city: string; }; } type PartialUser = DeepPartial<User>;

🎨 Decorators & Metadata

Master TypeScript decorators to add metadata, modify behavior, and implement cross-cutting concerns. Learn about class, method, property, and parameter decorators.

Creating Custom Decorators

// Enable experimental decorators in tsconfig.json // "experimentalDecorators": true, // "emitDecoratorMetadata": true // Class decorator function Entity<T extends { new(...args: any[]): {} }>( constructor: T ) { return class extends constructor { id = Math.random(); createdAt = new Date(); }; } @Entity class User { constructor(public name: string) {} } const user = new User("John"); console.log(user.id); // Random number console.log(user.createdAt); // Current date // Method decorator function log(target: any, propertyName: string, descriptor: PropertyDescriptor) { const method = descriptor.value; descriptor.value = function(...args: any[]) { console.log(`Calling ${propertyName} with args:`, args); const result = method.apply(this, args); console.log(`${propertyName} returned:`, result); return result; }; } class Calculator { @log add(a: number, b: number): number { return a + b; } } const calc = new Calculator(); calc.add(2, 3); // Logs method call and result // Property decorator function required(target: any, propertyName: string) { let value: any; const getter = () => value; const setter = (newValue: any) => { if (newValue === undefined || newValue === null) { throw new Error(`${propertyName} is required`); } value = newValue; }; Object.defineProperty(target, propertyName, { get: getter, set: setter, enumerable: true, configurable: true }); } class Product { @required name!: string; @required price!: number; } const product = new Product(); product.name = "Laptop"; // product.price = undefined; // Throws error // Parameter decorator function validate(target: any, methodName: string, parameterIndex: number) { // Store validation metadata const validators = Reflect.getMetadata("validators", target) || []; validators.push({ methodName, parameterIndex }); Reflect.defineMetadata("validators", validators, target); } class UserService { createUser(@validate name: string, @validate email: string): void { console.log(`Creating user: ${name}, ${email}`); } } // Decorator factory function cache(seconds: number = 60) { return function(target: any, propertyName: string, descriptor: PropertyDescriptor) { const method = descriptor.value; const cache = new Map(); descriptor.value = function(...args: any[]) { const key = JSON.stringify(args); if (cache.has(key)) { const cached = cache.get(key); if (Date.now() - cached.timestamp < seconds * 1000) { return cached.value; } } const result = method.apply(this, args); cache.set(key, { value: result, timestamp: Date.now() }); return result; }; }; } class ApiService { @cache(30) // Cache for 30 seconds async getUser(id: number): Promise<any> { // Simulate API call return { id, name: `User ${id}` }; } }

Advanced Decorator Patterns

// Authentication decorator function authenticate(roles: string[] = []) { return function(target: any, propertyName: string, descriptor: PropertyDescriptor) { const method = descriptor.value; descriptor.value = function(...args: any[]) { const currentUser = this.getCurrentUser(); // Assume this method exists if (!currentUser) { throw new Error("Authentication required"); } if (roles.length > 0 && !roles.includes(currentUser.role)) { throw new Error(`Access denied. Required roles: ${roles.join(", ")}`); } return method.apply(this, args); }; }; } class AdminController { @authenticate(["admin"]) deleteUser(userId: number): void { console.log(`Deleting user ${userId}`); } @authenticate(["admin", "moderator"]) banUser(userId: number): void { console.log(`Banning user ${userId}`); } private getCurrentUser() { return { id: 1, role: "admin" }; } } // Performance monitoring decorator function performanceMonitor(target: any, propertyName: string, descriptor: PropertyDescriptor) { const method = descriptor.value; descriptor.value = async function(...args: any[]) { const start = performance.now(); try { const result = await method.apply(this, args); const end = performance.now(); console.log(`${propertyName} took ${(end - start).toFixed(2)}ms`); return result; } catch (error) { const end = performance.now(); console.log(`${propertyName} failed after ${(end - start).toFixed(2)}ms`); throw error; } }; } class DatabaseService { @performanceMonitor async query(sql: string): Promise<any[]> { // Simulate database query await new Promise(resolve => setTimeout(resolve, 100)); return [{ id: 1, name: "Test" }]; } } // Validation decorator function validate(schema: any) { return function(target: any, propertyName: string, descriptor: PropertyDescriptor) { const method = descriptor.value; descriptor.value = function(...args: any[]) { const [data] = args; const { error } = schema.validate(data); if (error) { throw new Error(`Validation failed: ${error.details[0].message}`); } return method.apply(this, args); }; }; } // Example with Joi validation library const userSchema = { validate: (data: any) => { if (!data.name || data.name.length < 2) { return { error: { details: [{ message: "Name is required and must be at least 2 characters" }] } }; } return { error: null }; } }; class UserController { @validate(userSchema) createUser(userData: any): void { console.log("User created:", userData); } }
