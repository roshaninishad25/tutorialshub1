🛣️ App Router Deep Dive

The App Router is a revolutionary new routing system in Next.js 14 that provides improved performance, developer experience, and powerful new capabilities.

Basic Routing

// app/page.tsx - Home page export default function HomePage() { return ( <main> <h1>Welcome to Next.js 14</h1> <p>This is the home page using App Router.</p> </main> ); } // app/about/page.tsx - About page export default function AboutPage() { return ( <div> <h1>About Us</h1> <p>Learn more about our company.</p> </div> ); } // app/blog/[slug]/page.tsx - Dynamic route interface BlogPostPageProps { params: { slug: string; }; } export default function BlogPostPage({ params }: BlogPostPageProps) { return ( <article> <h1>Blog Post: {params.slug}</h1> <p>This is a dynamic blog post page.</p> </article> ); } // app/shop/[...slug]/page.tsx - Catch-all route interface ShopPageProps { params: { slug: string[]; }; } export default function ShopPage({ params }: ShopPageProps) { return ( <div> <h1>Shop</h1> <p>Path: {params.slug.join('/')}</p> </div> ); }

🖥️ Server Components

Server Components are a revolutionary feature that allows components to run on the server, providing direct access to data sources and improved performance.

Basic Server Components

// app/components/UserProfile.tsx - Server component import { getUser } from '@/lib/users'; interface UserProfileProps { userId: string; } export default async function UserProfile({ userId }: UserProfileProps) { const user = await getUser(userId); return ( <div className="user-profile"> <h2>{user.name}</h2> <p>{user.email}</p> <p>Member since: {new Date(user.createdAt).toLocaleDateString()}</p> </div> ); } // app/users/[id]/page.tsx - Using server component import UserProfile from '@/components/UserProfile'; interface UserPageProps { params: { id: string; }; } export default function UserPage({ params }: UserPageProps) { return ( <div> <h1>User Details</h1> <UserProfile userId={params.id} /> </div> ); } // app/components/PostList.tsx - Server component with data fetching import { getPosts } from '@/lib/posts'; export default async function PostList() { const posts = await getPosts(); return ( <div className="post-list"> <h2>Latest Posts</h2> {posts.map((post) => ( <article key={post.id} className="post-item"> <h3>{post.title}</h3> <p>{post.excerpt}</p> <small>{new Date(post.createdAt).toLocaleDateString()}</small> </article> ))} </div> ); }

Mixing Server and Client Components

const handleClick = () => { setCount(prev => prev + 1); onClick?.(); }; return ( <button onClick={handleClick}> {children} (clicked {count} times) </button> ); } // app/components/UserCard.tsx - Mixed component import { getUser } from '@/lib/users'; return ( <div className="user-card"> <h3>{user.name}</h3> <p>{user.bio}</p> <InteractiveButton> Follow {user.name} </InteractiveButton> </div> ); } // app/components/SearchableUserList.tsx - Server component with client interactivity import { searchUsers } from '@/lib/users'; import UserSearch from './UserSearch'; interface SearchableUserListProps { initialQuery?: string; } export default async function SearchableUserList({ initialQuery = '' }: SearchableUserListProps) { const users = await searchUsers(initialQuery); return ( <div> <h2>User Directory</h2> <UserSearch initialUsers={users} /> </div> ); } // app/components/UserSearch.tsx - Client component for search functionality 'use client'; import { useState, useEffect } from 'react'; import { searchUsers } from '@/lib/users'; interface User { id: string; name: string; email: string; } interface UserSearchProps { initialUsers: User[]; } export default function UserSearch({ initialUsers }: UserSearchProps) { const [users, setUsers] = useState(initialUsers); const [query, setQuery] = useState(''); const [loading, setLoading] = useState(false); useEffect(() => { if (query.trim()) { setLoading(true); searchUsers(query).then((results) => { setUsers(results); setLoading(false); }); } else { setUsers(initialUsers); } }, [query, initialUsers]); return ( <div> <input type="text" placeholder="Search users..." value={query} onChange={(e) => setQuery(e.target.value)} className="search-input" /> {loading ? ( <p>Searching...</p> ) : ( <ul> {users.map((user) => ( <li key={user.id}> {user.name} - {user.email} </li> ))} </ul> )} </div> ); }

// app/components/DataProvider.tsx - Data fetching pattern import { cache } from 'react'; interface Post { id: string; title: string; content: string; author: string; createdAt: string; } // Cached data fetching function const getPosts = cache(async () => { const response = await fetch('https://api.example.com/posts', { next: { revalidate: 3600 }, // Revalidate every hour }); return response.json(); }); export default async function DataProvider({ children, }: { children: React.ReactNode; }) { const posts = await getPosts(); return ( <PostContext.Provider value={posts}> {children} </PostContext.Provider> ); } // app/components/StreamingContent.tsx - Streaming pattern import { Suspense } from 'react'; import UserCard from './UserCard'; import PostList from './PostList'; export default function StreamingContent() { return ( <div> <h1>Dashboard</h1> <Suspense fallback={<div>Loading user info...</div>}> <UserCard userId="123" /> </Suspense> <Suspense fallback={<div>Loading posts...</div>}> <PostList /> </Suspense> </div> ); } // app/components/ErrorBoundary.tsx - Error handling for server components interface ErrorBoundaryProps { children: React.ReactNode; fallback?: React.ReactNode; } export default function ErrorBoundary({ children, fallback }: ErrorBoundaryProps) { return ( <div className="error-boundary"> {children} </div> ); } // app/error.tsx - Global error boundary 'use client'; import { useEffect } from 'react'; export default function Error({ error, reset, }: { error: Error & { digest?: string }; reset: () => void; }) { useEffect(() => { console.error(error); }, [error]); return ( <div> <h2>Something went wrong!</h2> <button onClick={reset}>Try again</button> </div> ); }

⚡ Server Actions

Server Actions allow you to define server-side functions that can be called from client components, simplifying data mutations and form handling.

Basic Server Actions

// app/actions/user.ts - Server action definitions 'use server'; import { revalidatePath } from 'next/cache'; import { redirect } from 'next/navigation'; import { z } from 'zod'; const CreateUserSchema = z.object({ name: z.string().min(2), email: z.string().email(), }); export async function createUser(formData: FormData) { const validatedFields = CreateUserSchema.safeParse({ name: formData.get('name'), email: formData.get('email'), }); if (!validatedFields.success) { return { errors: validatedFields.error.flatten().fieldErrors, }; } try { // Create user in database const response = await fetch('https://api.example.com/users', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(validatedFields.data), }); if (!response.ok) { throw new Error('Failed to create user'); } // Revalidate the users page cache revalidatePath('/users'); // Redirect to users page redirect('/users'); } catch (error) { return { error: 'Failed to create user. Please try again.', }; } } export async function updateUser(userId: string, formData: FormData) { const validatedFields = CreateUserSchema.safeParse({ name: formData.get('name'), email: formData.get('email'), }); if (!validatedFields.success) { return { errors: validatedFields.error.flatten().fieldErrors, }; } try { const response = await fetch(`https://api.example.com/users/${userId}`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(validatedFields.data), }); if (!response.ok) { throw new Error('Failed to update user'); } revalidatePath('/users'); revalidatePath(`/users/${userId}`); } catch (error) { return { error: 'Failed to update user. Please try again.', }; } } export async function deleteUser(userId: string) { try { const response = await fetch(`https://api.example.com/users/${userId}`, { method: 'DELETE', }); if (!response.ok) { throw new Error('Failed to delete user'); } revalidatePath('/users'); redirect('/users'); } catch (error) { return { error: 'Failed to delete user. Please try again.', }; } }

Using Server Actions in Forms

// app/users/create/page.tsx - Create user page import { createUser } from '@/app/actions/user'; export default function CreateUserPage() { return ( <div> <h1>Create New User</h1> <form action={createUser}> <div> <label htmlFor="name">Name:</label> <input type="text" id="name" name="name" required /> </div> <div> <label htmlFor="email">Email:</label> <input type="email" id="email" name="email" required /> </div> <button type="submit">Create User</button> </form> </div> ); } // app/components/UserForm.tsx - Reusable form component 'use client'; import { useFormState } from 'react-dom'; import { createUser } from '@/app/actions/user'; interface UserFormProps { initialData?: { name?: string; email?: string; }; } export default function UserForm({ initialData = {} }: UserFormProps) { const [state, formAction] = useFormState(createUser, null); return ( <form action={formAction}> <div> <label htmlFor="name">Name:</label> <input type="text" id="name" name="name" defaultValue={initialData.name} required /> {state?.errors?.name && ( <p className="error">{state.errors.name[0]}</p> )} </div> <div> <label htmlFor="email">Email:</label> <input type="email" id="email" name="email" defaultValue={initialData.email} required /> {state?.errors?.email && ( <p className="error">{state.errors.email[0]}</p> )} </div> {state?.error && ( <p className="error">{state.error}</p> )} <button type="submit"> {initialData.name ? 'Update User' : 'Create User'} </button> </form> ); }

Advanced Server Action Patterns

// app/actions/post.ts - Advanced server actions 'use server'; import { revalidateTag, revalidatePath } from 'next/cache'; import { headers } from 'next/headers'; export async function createPost(formData: FormData) { // Get headers for authentication const headersList = headers(); const authorization = headersList.get('authorization'); if (!authorization) { return { error: 'Authentication required', }; } // Validate user const user = await validateUser(authorization); if (!user) { return { error: 'Invalid authentication', }; } const title = formData.get('title') as string; const content = formData.get('content') as string; if (!title || !content) { return { error: 'Title and content are required', }; } try { const response = await fetch('https://api.example.com/posts', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': authorization, }, body: JSON.stringify({ title, content, authorId: user.id, }), }); if (!response.ok) { throw new Error('Failed to create post'); } const post = await response.json(); // Revalidate specific cache tags revalidateTag('posts'); revalidateTag(`user-${user.id}-posts`); // Revalidate specific paths revalidatePath('/posts'); revalidatePath('/dashboard'); return { success: true, post, }; } catch (error) { return { error: 'Failed to create post. Please try again.', }; } } // Server action with file upload export async function uploadImage(formData: FormData) { const file = formData.get('image') as File; if (!file) { return { error: 'No file provided', }; } // Validate file type if (!file.type.startsWith('image/')) { return { error: 'File must be an image', }; } // Validate file size (5MB max) if (file.size > 5 * 1024 * 1024) { return { error: 'File size must be less than 5MB', }; } try { const formData = new FormData(); formData.append('image', file); const response = await fetch('https://api.example.com/upload', { method: 'POST', body: formData, }); if (!response.ok) { throw new Error('Failed to upload image'); } const result = await response.json(); return { success: true, imageUrl: result.url, }; } catch (error) { return { error: 'Failed to upload image. Please try again.', }; } } // Server action with optimistic updates export async function toggleLike(postId: string, userId: string) { try { const response = await fetch(`https://api.example.com/posts/${postId}/like`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ userId }), }); if (!response.ok) { throw new Error('Failed to toggle like'); } const result = await response.json(); // Revalidate specific post cache revalidateTag(`post-${postId}`); revalidateTag('user-likes'); return { success: true, liked: result.liked, likesCount: result.likesCount, }; } catch (error) { return { error: 'Failed to toggle like. Please try again.', }; } } // Helper function for user validation async function validateUser(authorization: string) { try { const response = await fetch('https://api.example.com/auth/me', { headers: { Authorization: authorization }, }); if (!response.ok) { return null; } return response.json(); } catch { return null; } }
