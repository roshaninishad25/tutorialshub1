# Microservices Architecture Example # User Service (Node.js + Express) # user-service/package.json { "name": "user-service", "version": "1.0.0", "description": "User Management Microservice", "main": "server.js", "dependencies": { "express": "^4.18.2", "mongoose": "^7.0.0", "jsonwebtoken": "^9.0.0", "bcryptjs": "^2.4.3", "cors": "^2.8.5", "helmet": "^6.0.1", "dotenv": "^16.0.3", "winston": "^3.8.2", "prom-client": "^14.2.0", "circuit-breaker-js": "^1.0.1" } } # user-service/server.js const express = require('express'); const mongoose = require('mongoose'); const cors = require('cors'); const helmet = require('helmet'); require('dotenv').config(); const userService = require('./services/userService'); const authMiddleware = require('./middleware/auth'); const rateLimiter = require('./middleware/rateLimiter'); const circuitBreaker = require('./middleware/circuitBreaker'); const logger = require('./utils/logger'); const metrics = require('./utils/metrics'); const app = express(); // Security middleware app.use(helmet()); app.use(cors({ origin: process.env.ALLOWED_ORIGINS?.split(',') || ['http://localhost:3000'], credentials: true })); // Rate limiting app.use(rateLimiter); // Metrics middleware app.use(metrics.middleware); // Body parsing app.use(express.json()); // Database connection mongoose.connect(process.env.MONGODB_URI || 'mongodb://localhost:27017/user-service', { useNewUrlParser: true, useUnifiedTopology: true, }) .then(() => logger.info('Connected to MongoDB')) .catch(err => logger.error('MongoDB connection error:', err)); // Health check app.get('/health', (req, res) => { res.status(200).json({ service: 'user-service', status: 'healthy', timestamp: new Date().toISOString(), version: process.env.npm_package_version }); }); // Metrics endpoint app.get('/metrics', (req, res) => { res.set('Content-Type', metrics.register.contentType); res.end(metrics.register.metrics()); }); // Routes app.post('/api/users', circuitBreaker(userService.createUser)); app.get('/api/users/:id', authMiddleware, circuitBreaker(userService.getUserById)); app.put('/api/users/:id', authMiddleware, circuitBreaker(userService.updateUser)); app.delete('/api/users/:id', authMiddleware, circuitBreaker(userService.deleteUser)); app.post('/api/auth/login', circuitBreaker(userService.login)); app.post('/api/auth/refresh', circuitBreaker(userService.refreshToken)); // Error handling app.use((err, req, res, next) => { logger.error('Unhandled error:', err); res.status(500).json({ error: 'Internal Server Error', message: process.env.NODE_ENV === 'development' ? err.message : 'Something went wrong' }); }); const PORT = process.env.PORT || 3001; app.listen(PORT, () => { logger.info(`User service running on port ${PORT}`); }); # user-service/services/userService.js const User = require('../models/User'); const jwt = require('jsonwebtoken'); const bcrypt = require('bcryptjs'); const logger = require('../utils/logger'); const eventBus = require('../utils/eventBus'); class UserService { async createUser(userData) { try { const { email, username, password, firstName, lastName } = userData; // Check if user already exists const existingUser = await User.findOne({ $or: [{ email }, { username }] }); if (existingUser) { throw new Error('User with this email or username already exists'); } // Hash password const hashedPassword = await bcrypt.hash(password, 12); const user = new User({ email, username, password: hashedPassword, firstName, lastName }); await user.save(); // Publish user created event await eventBus.publish('user.created', { userId: user._id, email: user.email, username: user.username, firstName: user.firstName, lastName: user.lastName, timestamp: new Date().toISOString() }); logger.info(`User created: ${user._id}`); return { id: user._id, email: user.email, username: user.username, firstName: user.firstName, lastName: user.lastName }; } catch (error) { logger.error('Error creating user:', error); throw error; } } async getUserById(userId) { try { const user = await User.findById(userId).select('-password'); if (!user) { throw new Error('User not found'); } return user; } catch (error) { logger.error('Error getting user:', error); throw error; } } async updateUser(userId, updateData) { try { const user = await User.findByIdAndUpdate( userId, updateData, { new: true, runValidators: true } ).select('-password'); if (!user) { throw new Error('User not found'); } // Publish user updated event await eventBus.publish('user.updated', { userId: user._id, updatedFields: Object.keys(updateData), timestamp: new Date().toISOString() }); logger.info(`User updated: ${userId}`); return user; } catch (error) { logger.error('Error updating user:', error); throw error; } } async deleteUser(userId) { try { const user = await User.findByIdAndDelete(userId); if (!user) { throw new Error('User not found'); } // Publish user deleted event await eventBus.publish('user.deleted', { userId: user._id, email: user.email, timestamp: new Date().toISOString() }); logger.info(`User deleted: ${userId}`); return { message: 'User deleted successfully' }; } catch (error) { logger.error('Error deleting user:', error); throw error; } } async login(email, password) { try { const user = await User.findOne({ email, isActive: true }); if (!user) { throw new Error('Invalid credentials'); } const isPasswordValid = await bcrypt.compare(password, user.password); if (!isPasswordValid) { throw new Error('Invalid credentials'); } // Generate tokens const accessToken = jwt.sign( { userId: user._id, email: user.email }, process.env.JWT_SECRET, { expiresIn: '15m' } ); const refreshToken = jwt.sign( { userId: user._id }, process.env.JWT_REFRESH_SECRET, { expiresIn: '7d' } ); // Update last login user.lastLogin = new Date(); await user.save(); // Publish login event await eventBus.publish('user.login', { userId: user._id, timestamp: new Date().toISOString() }); logger.info(`User logged in: ${user._id}`); return { accessToken, refreshToken, user: { id: user._id, email: user.email, username: user.username, firstName: user.firstName, lastName: user.lastName } }; } catch (error) { logger.error('Error during login:', error); throw error; } } async refreshToken(refreshToken) { try { const decoded = jwt.verify(refreshToken, process.env.JWT_REFRESH_SECRET); const user = await User.findById(decoded.userId).select('-password'); if (!user || !user.isActive) { throw new Error('Invalid refresh token'); } const newAccessToken = jwt.sign( { userId: user._id, email: user.email }, process.env.JWT_SECRET, { expiresIn: '15m' } ); return { accessToken: newAccessToken }; } catch (error) { logger.error('Error refreshing token:', error); throw error; } } } module.exports = new UserService(); # API Gateway Configuration # api-gateway/kong.yml _format_version: "3.0" services: - name: user-service url: http://user-service:3001 plugins: - name: rate-limiting config: minute: 100 hour: 1000 - name: prometheus config: per_consumer: true routes: - name: user-routes paths: - /api/users - /api/auth methods: - GET - POST - PUT - DELETE - name: order-service url: http://order-service:3002 plugins: - name: rate-limiting config: minute: 200 hour: 2000 - name: prometheus config: per_consumer: true routes: - name: order-routes paths: - /api/orders methods: - GET - POST - PUT - DELETE - name: product-service url: http://product-service:3003 plugins: - name: rate-limiting config: minute: 500 hour: 5000 - name: prometheus config: per_consumer: true routes: - name: product-routes paths: - /api/products methods: - GET - POST - PUT - DELETE # Docker Compose for Microservices # docker-compose.yml version: '3.8' services: # API Gateway kong: image: kong:3.3 environment: KONG_DATABASE: "off" KONG_DECLARATIVE_CONFIG: /kong/declarative/kong.yml Best Practices

  • ✅ Choose the right architecture pattern for your use case
  • ✅ Implement proper service boundaries and responsibilities
  • ✅ Design for scalability and maintainability from the start
  • ✅ Implement proper monitoring and observability
  • ✅ Use circuit breakers and fallback mechanisms
  • ✅ Implement proper data consistency strategies
  • ✅ Design for failure and implement resilience patterns
  • ✅ Document architecture decisions and trade-offs
