Firewall Configuration

# Basic iptables firewall configuration # Clear existing rules iptables -F iptables -X iptables -t nat -F iptables -t nat -X # Set default policies iptables -P INPUT DROP iptables -P FORWARD DROP iptables -P OUTPUT ACCEPT # Allow loopback traffic iptables -A INPUT -i lo -j ACCEPT # Allow established and related connections iptables -A INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT # Allow SSH (port 22) iptables -A INPUT -p tcp --dport 22 -j ACCEPT # Allow HTTP (port 80) iptables -A INPUT -p tcp --dport 80 -j ACCEPT # Allow HTTPS (port 443) iptables -A INPUT -p tcp --dport 443 -j ACCEPT # Rate limiting for SSH to prevent brute force attacks iptables -A INPUT -p tcp --dport 22 -m limit --limit 3/min --limit-burst 3 -j ACCEPT iptables -A INPUT -p tcp --dport 22 -j DROP # Log dropped packets iptables -A INPUT -j LOG --log-prefix "Dropped: " # Save rules iptables-save > /etc/iptables/rules.v4 # UFW (Uncomplicated Firewall) - Ubuntu/Debian # Enable UFW ufw enable # Allow specific ports ufw allow 22/tcp # SSH ufw allow 80/tcp # HTTP ufw allow 443/tcp # HTTPS # Deny specific IP ufw deny from 192.168.1.100 # Allow from specific subnet ufw allow from 192.168.1.0/24 # Show status ufw status verbose ufw status numbered # Windows Firewall with PowerShell # Enable firewall Set-NetFirewallProfile -Profile Domain,Public,Private -Enabled True # Block specific IP New-NetFirewallRule -DisplayName "Block Malicious IP" -Direction Inbound -RemoteAddress 192.168.1.100 -Action Block # Allow specific port New-NetFirewallRule -DisplayName "Allow HTTP" -Direction Inbound -Protocol TCP -LocalPort 80 -Action Allow # Show firewall rules Get-NetFirewallRule | Format-Table DisplayName, Enabled, Direction, Action, Protocol, LocalPort

Intrusion Detection Systems

# Snort IDS Configuration # Install Snort sudo apt-get install snort # Basic snort.conf configuration # /etc/snort/snort.conf # Set network variables var HOME_NET 192.168.1.0/24 var EXTERNAL_NET !$HOME_NET # Configure detection rules # Alert on port scans alert tcp $EXTERNAL_NET any -> $HOME_NET any (msg:"Port Scan Detected"; flags:S; threshold:type both, track by_src, count 5, seconds 10; sid:1000001;) # Alert on suspicious HTTP traffic alert http $EXTERNAL_NET any -> $HOME_NET 80 (msg:"Suspicious HTTP User Agent"; content:"User-Agent: sqlmap"; nocase; sid:1000002;) # Alert on brute force attempts alert tcp $EXTERNAL_NET any -> $HOME_NET 22 (msg:"SSH Brute Force Attempt"; threshold:type both, track by_src, count 5, seconds 60; sid:1000003;) # Run Snort in IDS mode sudo snort -A console -i eth0 -c /etc/snort/snort.conf # Suricata - Modern IDS/IPS # Install Suricata sudo apt-get install suricata # Basic suricata.yaml configuration # /etc/suricata/suricata.yaml # Configure network interfaces af-packet: - interface: eth0 cluster-id: 99 cluster-type: cluster_flow # Configure logging outputs: - fast: enabled: yes filename: fast.log - eve-log: enabled: yes type: file filename: eve.json # Run Suricata sudo suricata -c /etc/suricata/suricata.yaml -i eth0 # Monitor logs tail -f /var/log/suricata/fast.log tail -f /var/log/suricata/eve.json | jq .

Best Practices

  • ✅ Follow the principle of least privilege
  • ✅ Implement defense in depth strategies
  • ✅ Keep systems and software updated
  • ✅ Use strong, unique passwords and multi-factor authentication
  • ✅ Regularly backup critical data
  • ✅ Monitor and log security events
  • ✅ Conduct regular security assessments
  • ✅ Educate users about security best practices
