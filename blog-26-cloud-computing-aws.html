AWS CLI and SDK Setup

# Install AWS CLI # macOS brew install awscli # Ubuntu/Debian sudo apt-get update sudo apt-get install awscli # Windows # Download and run the installer from aws.amazon.com/cli # Configure AWS CLI aws configure # Enter your credentials when prompted: # AWS Access Key ID: YOUR_ACCESS_KEY # AWS Secret Access Key: YOUR_SECRET_KEY # Default region name: us-west-2 # Default output format: json # Verify configuration aws sts get-caller-identity # List available services aws help # Working with S3 via CLI # List buckets aws s3 ls # Create bucket aws s3 mb s3://my-unique-bucket-name # Upload file aws s3 cp myfile.txt s3://my-unique-bucket-name/ # Download file aws s3 cp s3://my-unique-bucket-name/myfile.txt ./myfile.txt # Sync directory aws s3 sync ./local-folder s3://my-unique-bucket-name/remote-folder # Working with EC2 via CLI # List instances aws ec2 describe-instances # List available instance types aws ec2 describe-instance-types --query "InstanceTypes[0:5].{Type:InstanceType,VCPU:VCpuInfo.DefaultVCpus,Memory:MemoryInfo.SizeInMiB}" # Create security group aws ec2 create-security-group --group-name my-sg --description "My security group" # Authorize inbound traffic aws ec2 authorize-security-group-ingress --group-id sg-xxxxxxxx --protocol tcp --port 22 --cidr 0.0.0.0/0 # Working with IAM via CLI # Create user aws iam create-user --user-name myuser # Create access key aws iam create-access-key --user-name myuser # List users aws iam list-users # Create policy aws iam create-policy --policy-name MyPolicy --policy-document file://policy.json # Attach policy to user aws iam attach-user-policy --user-name myuser --policy-arn arn:aws:iam::account:policy/MyPolicy # AWS SDK for Python (Boto3) example import boto3 import json # Create S3 client s3 = boto3.client('s3') # List buckets response = s3.list_buckets() print("Buckets:") for bucket in response['Buckets']: print(f" {bucket['Name']}") # Create bucket s3.create_bucket(Bucket='my-python-bucket') # Upload file s3.upload_file('localfile.txt', 'my-python-bucket', 'remotefile.txt') # Download file s3.download_file('my-python-bucket', 'remotefile.txt', 'downloaded.txt') # Working with EC2 ec2 = boto3.client('ec2') # List instances response = ec2.describe_instances() print("EC2 Instances:") for reservation in response['Reservations']: for instance in reservation['Instances']: print(f" Instance ID: {instance['InstanceId']}") print(f" State: {instance['State']['Name']}") # Create EC2 instance response = ec2.run_instances( ImageId='ami-0c55b159cbfafe1f0', # Amazon Linux 2 MinCount=1, MaxCount=1, InstanceType='t2.micro', KeyName='my-key-pair', SecurityGroupIds=['sg-xxxxxxxx'] ) instance_id = response['Instances'][0]['InstanceId'] print(f"Created instance: {instance_id}")

AWS Infrastructure and Regions

# AWS Global Infrastructure # Regions: Geographic areas where AWS has data centers # Availability Zones: Isolated locations within regions # Edge Locations: Sites for CloudFront content delivery # List all regions aws ec2 describe-regions --query "Regions[].{Name:RegionName}" # List availability zones for a region aws ec2 describe-availability-zones --region us-west-2 # Working with multiple regions export AWS_DEFAULT_REGION=us-west-2 aws s3 ls # Or specify region per command aws s3 ls --region eu-west-1 # AWS CloudFormation - Infrastructure as Code # template.yaml AWSTemplateFormatVersion: '2010-09-09' Description: 'Simple EC2 instance with S3 bucket' Parameters: InstanceType: Type: String Default: t2.micro Description: EC2 instance type KeyName: Type: AWS::EC2::KeyPair::KeyName Description: Name of an existing EC2 KeyPair Resources: MyEC2Instance: Type: AWS::EC2::Instance Properties: InstanceType: !Ref InstanceType KeyName: !Ref KeyName ImageId: ami-0c55b159cbfafe1f0 SecurityGroupIds: - !Ref MySecurityGroup MySecurityGroup: Type: AWS::EC2::SecurityGroup Properties: GroupDescription: Enable SSH access SecurityGroupIngress: - IpProtocol: tcp FromPort: 22 ToPort: 22 CidrIp: 0.0.0.0/0 MyS3Bucket: Type: AWS::S3::Bucket Properties: BucketName: !Sub '${AWS::StackName}-bucket-${AWS::AccountId}' Outputs: InstanceId: Description: Instance ID of the EC2 instance Value: !Ref MyEC2Instance BucketName: Description: Name of the S3 bucket Value: !Ref MyS3Bucket # Deploy CloudFormation stack aws cloudformation create-stack \ --stack-name my-stack \ --template-body file://template.yaml \ --parameters ParameterKey=KeyName,ParameterValue=my-key-pair # Monitor stack creation aws cloudformation describe-stacks --stack-name my-stack # Delete stack aws cloudformation delete-stack --stack-name my-stack # AWS CDK (Cloud Development Kit) - Infrastructure as Code with TypeScript # Install CDK npm install -g aws-cdk # Initialize CDK project mkdir my-cdk-app cd my-cdk-app cdk init app --language typescript # lib/my-cdk-app-stack.ts import * as cdk from 'aws-cdk-lib'; import { Construct } from 'constructs'; import * as ec2 from 'aws-cdk-lib/aws-ec2'; import * as s3 from 'aws-cdk-lib/aws-s3'; export class MyCdkAppStack extends cdk.Stack { constructor(scope: Construct, id: string, props?: cdk.StackProps) { super(scope, id, props); // Create VPC const vpc = new ec2.Vpc(this, 'MyVpc', { cidr: '10.0.0.0/16', natGateways: 1, maxAzs: 2 }); // Create security group const securityGroup = new ec2.SecurityGroup(this, 'MySecurityGroup', { vpc, allowAllOutbound: true, description: 'Allow SSH access' }); securityGroup.addIngressRule( ec2.Peer.anyIpv4(), ec2.Port.tcp(22), 'Allow SSH access' ); // Create EC2 instance const instance = new ec2.Instance(this, 'MyInstance', { vpc, instanceType: ec2.InstanceType.of(ec2.InstanceClass.T2, ec2.InstanceSize.MICRO), machineImage: ec2.MachineImage.latestAmazonLinux(), securityGroup, vpcSubnets: { subnetType: ec2.SubnetType.PUBLIC } }); // Create S3 bucket const bucket = new s3.Bucket(this, 'MyBucket', { versioned: true, encryption: s3.BucketEncryption.S3_MANAGED, blockPublicAccess: s3.BlockPublicAccess.BLOCK_ALL }); // Output values new cdk.CfnOutput(this, 'InstanceId', { value: instance.instanceId }); new cdk.CfnOutput(this, 'BucketName', { value: bucket.bucketName }); } } # Deploy CDK stack cdk deploy # Destroy CDK stack cdk destroy

Best Practices

  • ✅ Use Infrastructure as Code for reproducible deployments
  • ✅ Implement proper IAM policies and principle of least privilege
  • ✅ Enable monitoring and logging for all resources
  • ✅ Use multiple Availability Zones for high availability
  • ✅ Implement proper backup and disaster recovery strategies
  • ✅ Optimize costs using appropriate instance types and reserved capacity
  • ✅ Use security groups and network ACLs for network security
  • ✅ Regularly review and update security configurations
