🔍 Introduction to GraphQL

GraphQL is a query language for APIs and a runtime for executing those queries with your existing data. It provides a more efficient, powerful, and flexible alternative to REST for building modern API-driven applications.

This comprehensive course teaches frontend developers to master GraphQL, from basic queries to advanced caching strategies, real-time subscriptions, and production-ready implementations with Apollo Client.

📋 GraphQL Queries Deep Dive

Master GraphQL queries to fetch exactly the data you need. Learn about fields, arguments, variables, fragments, aliases, and advanced query patterns.

Basic Query Structure

# Basic GraphQL query query GetUser { user(id: "1") { id name email avatar } } # Query with arguments query GetPosts($limit: Int!, $offset: Int!) { posts(limit: $limit, offset: $offset) { id title content author { name avatar } createdAt } } # Query variables { "limit": 10, "offset": 0 } # Query with aliases query GetUserAndPosts { user: user(id: "1") { id name email } posts: posts(authorId: "1") { id title content } } # Query with fragments fragment UserFields on User { id name email avatar } query GetUserWithFragment { user(id: "1") { ...UserFields posts { id title } } }

Advanced Query Patterns

# Nested queries with relationships query GetPostWithComments { post(id: "1") { id title content author { id name avatar posts { id title } } comments { id content author { id name avatar } replies { id content author { id name } } } } } # Query with directives query GetPostsWithConditionalFields { posts { id title content @include(if: $showContent) author @skip(if: $hideAuthor) { id name } createdAt @deprecated(reason: "Use publishedAt instead") } } # Inline fragments for union types query SearchResults($query: String!) { search(query: $query) { ... on Post { id title content } ... on User { id name avatar } ... on Comment { id content author { name } } } } # Query with input types query FilterPosts($filter: PostFilter!) { posts(filter: $filter) { id title content author { name } } } # Input type variables { "filter": { "authorId": "1", "publishedAfter": "2024-01-01", "tags": ["javascript", "react"], "limit": 10 } }

React Query Implementation

import { useQuery, gql } from '@apollo/client'; import React from 'react'; // GraphQL query definition const GET_USER_POSTS = gql` query GetUserPosts($userId: ID!, $limit: Int) { user(id: $userId) { id name email avatar posts(limit: $limit) { id title content publishedAt comments { id content } } } } `; // Component using the query const UserProfile = ({ userId }) => { const { loading, error, data, refetch, fetchMore } = useQuery(GET_USER_POSTS, { variables: { userId, limit: 5 }, notifyOnNetworkStatusChange: true, }); if (loading && !data) return <div>Loading...</div>; if (error) return <div>Error: {error.message}</div>; const { user } = data; const handleLoadMore = () => { fetchMore({ variables: { userId, limit: 5, offset: user.posts.length, }, updateQuery: (prev, { fetchMoreResult }) => { if (!fetchMoreResult) return prev; return { user: { ...prev.user, posts: [...prev.user.posts, ...fetchMoreResult.user.posts], }, }; }, }); }; return ( <div> <div className="user-header"> <img src={user.avatar} alt={user.name} /> <h2>{user.name}</h2> <p>{user.email}</p> </div> <div className="posts"> <h3>Posts ({user.posts.length})</h3> {user.posts.map((post) => ( <div key={post.id} className="post"> <h4>{post.title}</h4> <p>{post.content}</p> <small>{new Date(post.publishedAt).toLocaleDateString()}</small> <div className="comments"> {post.comments.length} comments </div> </div> ))} <button onClick={handleLoadMore} disabled={loading}> {loading ? 'Loading...' : 'Load More'} </button> </div> <button onClick={() => refetch()}> Refresh </button> </div> ); }; export default UserProfile;

✏️ Mutations & Data Updates

Learn to create, update, and delete data using GraphQL mutations. Master mutation design patterns, optimistic updates, and error handling.

Basic Mutations

# Create user mutation mutation CreateUser($input: CreateUserInput!) { createUser(input: $input) { id name email avatar createdAt } } # Mutation variables { "input": { "name": "John Doe", "email": "john@example.com", "password": "securepassword" } } # Update user mutation mutation UpdateUser($id: ID!, $input: UpdateUserInput!) { updateUser(id: $id, input: $input) { id name email avatar updatedAt } } # Delete user mutation mutation DeleteUser($id: ID!) { deleteUser(id: $id) { id success message } } # Create post mutation mutation CreatePost($input: CreatePostInput!) { createPost(input: $input) { id title content author { id name } createdAt } } # Multiple mutations in one request mutation CreateUserAndPost($userInput: CreateUserInput!, $postInput: CreatePostInput!) { createUser(input: $userInput) { id name email } createPost(input: $postInput) { id title content } }

React Mutation Implementation

import { useMutation, gql } from '@apollo/client'; import React, { useState } from 'react'; // Mutation definitions const CREATE_POST = gql` mutation CreatePost($input: CreatePostInput!) { createPost(input: $input) { id title content author { id name } createdAt } } `; const UPDATE_POST = gql` mutation UpdatePost($id: ID!, $input: UpdatePostInput!) { updatePost(id: $id, input: $input) { id title content updatedAt } } `; const DELETE_POST = gql` mutation DeletePost($id: ID!) { deletePost(id: $id) { id success message } } `; // Post form component const PostForm = ({ postId, initialData, onSave, onCancel }) => { const [title, setTitle] = useState(initialData?.title || ''); const [content, setContent] = useState(initialData?.content || ''); const isEdit = !!postId; const mutation = isEdit ? UPDATE_POST : CREATE_POST; const mutationName = isEdit ? 'updatePost' : 'createPost'; const [executeMutation, { loading, error }] = useMutation(mutation, { onCompleted: (data) => { onSave?.(data[mutationName]); }, onError: (error) => { console.error('Mutation error:', error); }, // Optimistic update optimisticResponse: isEdit ? { updatePost: { __typename: 'Post', id: postId, title, content, updatedAt: new Date().toISOString(), }, } : { createPost: { __typename: 'Post', id: 'temp-id', title, content, author: { __typename: 'User', id: '1', name: 'Current User' }, createdAt: new Date().toISOString(), }, }, // Update cache after mutation update: (cache, { data }) => { const newPost = data[mutationName]; if (isEdit) { // Update existing post in cache cache.modify({ id: cache.identify(newPost), fields: { title() { return newPost.title; }, content() { return newPost.content; }, updatedAt() { return newPost.updatedAt; }, }, }); } else { // Add new post to cache const existingPosts = cache.readQuery({ query: GET_USER_POSTS, variables: { userId: '1' }, }); if (existingPosts) { cache.writeQuery({ query: GET_USER_POSTS, variables: { userId: '1' }, data: { user: { ...existingPosts.user, posts: [newPost, ...existingPosts.user.posts], }, }, }); } } }, }); const handleSubmit = (e) => { e.preventDefault(); const variables = isEdit ? { id: postId, input: { title, content } } : { input: { title, content, authorId: '1' } }; executeMutation({ variables }); }; return ( <form onSubmit={handleSubmit} className="post-form"> <div className="form-group"> <label htmlFor="title">Title</label> <input type="text" id="title" value={title} onChange={(e) => setTitle(e.target.value)} required /> </div> <div className="form-group"> <label htmlFor="content">Content</label> <textarea id="content" value={content} onChange={(e) => setContent(e.target.value)} rows={5} required /> </div> {error && ( <div className="error"> Error: {error.message} </div> )} <div className="form-actions"> <button type="button" onClick={onCancel}> Cancel </button> <button type="submit" disabled={loading}> {loading ? 'Saving...' : (isEdit ? 'Update' : 'Create')} </button> </div> </form> ); }; // Post list with delete functionality const PostList = ({ posts, onDelete }) => { const [deletePost, { loading: deleting }] = useMutation(DELETE_POST, { onCompleted: (data) => { onDelete?.(data.deletePost); }, update: (cache, { data: { deletePost } }) => { if (deletePost.success) { cache.modify({ fields: { posts(existingPosts = [], { readField }) { return existingPosts.filter( postRef => deletePost.id !== readField('id', postRef) ); }, }, }); } }, }); const handleDelete = (postId) => { if (window.confirm('Are you sure you want to delete this post?')) { deletePost({ variables: { id: postId } }); } }; return ( <div className="post-list"> {posts.map((post) => ( <div key={post.id} className="post-item"> <h3>{post.title}</h3> <p>{post.content}</p> <div className="post-actions"> <button onClick={() => onEdit(post)}> Edit </button> <button onClick={() => handleDelete(post.id)} disabled={deleting} > {deleting ? 'Deleting...' : 'Delete'} </button> </div> </div> ))} </div> ); }; export { PostForm, PostList };

🚀 Apollo Client Integration

Master Apollo Client, the most popular GraphQL client for JavaScript applications. Learn setup, configuration, caching, and advanced features.

Apollo Client Setup

// apollo-client.js import { ApolloClient, InMemoryCache, createHttpLink, from } from '@apollo/client'; import { setContext } from '@apollo/client/link/context'; import { onError } from '@apollo/client/link/error'; import { RetryLink } from '@apollo/client/link/retry'; // HTTP link const httpLink = createHttpLink({ uri: process.env.REACT_APP_GRAPHQL_ENDPOINT || 'https://api.example.com/graphql', credentials: 'include', }); // Auth link for adding headers const authLink = setContext((_, { headers }) => { const token = localStorage.getItem('authToken'); return { headers: { ...headers, authorization: token ? `Bearer ${token}` : '', }, }; }); // Error handling link const errorLink = onError(({ graphQLErrors, networkError, operation, forward }) => { if (graphQLErrors) { graphQLErrors.forEach(({ message, locations, path }) => { console.error(`[GraphQL error]: Message: ${message}, Location: ${locations}, Path: ${path}`); }); // Handle authentication errors if (graphQLErrors.some(error => error.message === 'Authentication required')) { localStorage.removeItem('authToken'); window.location.href = '/login'; } } if (networkError) { console.error(`[Network error]: ${networkError}`); // Handle network errors if (networkError.statusCode === 401) { localStorage.removeItem('authToken'); window.location.href = '/login'; } } }); // Retry link for failed requests const retryLink = new RetryLink({ delay: { initial: 300, max: Infinity, jitter: true, }, attempts: { max: 3, retryIf: (error, _operation) => !!error, }, }); // Cache configuration const cache = new InMemoryCache({ typePolicies: { Query: { fields: { posts: { merge(existing = [], incoming) { return incoming; }, }, user: { merge(existing, incoming) { return { ...existing, ...incoming }; }, }, }, }, User: { fields: { posts: { merge(existing = [], incoming) { return incoming; }, }, }, }, Post: { fields: { comments: { merge(existing = [], incoming) { return incoming; }, }, }, }, }, }); // Apollo Client instance export const client = new ApolloClient({ link: from([retryLink, errorLink, authLink, httpLink]), cache, defaultOptions: { watchQuery: { errorPolicy: 'all', notifyOnNetworkStatusChange: true, }, query: { errorPolicy: 'all', }, mutate: { errorPolicy: 'all', }, }, }); // App.js with Apollo Provider import React from 'react'; import { ApolloProvider } from '@apollo/client'; import { client } from './apollo-client'; import { BrowserRouter as Router, Routes, Route } from 'react-router-dom'; import { ThemeProvider } from 'styled-components'; import { GlobalStyle } from './styles/GlobalStyle'; import { theme } from './styles/theme'; import { AuthProvider } from './contexts/AuthContext'; import { Layout } from './components/Layout'; import { Home } from './pages/Home'; import { Login } from './pages/Login'; import { Register } from './pages/Register'; import { Profile } from './pages/Profile'; import { ProtectedRoute } from './components/ProtectedRoute'; const App = () => { return ( <ApolloProvider client={client}> <ThemeProvider theme={theme}> <GlobalStyle /> <AuthProvider> <Router> <Routes> <Route path="/" element={<Layout><Home /></Layout>} /> <Route path="/login" element={<Login />} /> <Route path="/register" element={<Register />} /> <Route path="/profile" element={ <ProtectedRoute> <Layout> <Profile /> </Layout> </ProtectedRoute> } /> </Routes> </Router> </AuthProvider> </ThemeProvider> </ApolloProvider> ); }; export default App;

Advanced Apollo Features

// Custom hooks for common operations import { useQuery, useMutation, useSubscription } from '@apollo/client'; import { useCallback, useMemo } from 'react'; // Hook for paginated queries export const usePaginatedQuery = (query, options = {}) => { const { data, loading, error, fetchMore, refetch } = useQuery(query, { ...options, notifyOnNetworkStatusChange: true, }); const loadMore = useCallback(() => { if (!data || !fetchMore) return; const { pageInfo } = data; if (!pageInfo || !pageInfo.hasNextPage) return; return fetchMore({ variables: { ...options.variables, after: pageInfo.endCursor, }, updateQuery: (previousResult, { fetchMoreResult }) => { if (!fetchMoreResult) return previousResult; const edges = [ ...previousResult[Object.keys(previousResult)[0]].edges, ...fetchMoreResult[Object.keys(fetchMoreResult)[0]].edges, ]; return { ...fetchMoreResult, [Object.keys(fetchMoreResult)[0]]: { ...fetchMoreResult[Object.keys(fetchMoreResult)[0]], edges, }, }; }, }); }, [data, fetchMore, options.variables]); return { data, loading, error, loadMore, refetch, hasNextPage: data?.pageInfo?.hasNextPage || false, }; }; // Hook for optimistic mutations export const useOptimisticMutation = (mutation, options = {}) => { const [executeMutation, { loading, error, data }] = useMutation(mutation, { ...options, optimisticResponse: options.optimisticResponse, update: options.update, }); const mutate = useCallback((variables, optimisticResponse) => { return executeMutation({ variables, optimisticResponse: optimisticResponse || options.optimisticResponse, }); }, [executeMutation, options.optimisticResponse]); return { mutate, loading, error, data }; }; // Hook for local state management export const useLocalState = (defaultValue) => { const [state, setState] = useQuery(GET_LOCAL_STATE); const updateState = useCallback((newValue) => { setState({ variables: { value: newValue }, }); }, [setState]); return { value: state?.localState?.value || defaultValue, setValue: updateState, }; }; // Local state query const GET_LOCAL_STATE = gql` query GetLocalState { localState @client { value } } `; // Local state mutation const UPDATE_LOCAL_STATE = gql` mutation UpdateLocalState($value: String!) { updateLocalState(value: $value) @client } `; // Hook for subscription with fallback export const useSubscriptionWithFallback = (subscription, query, options = {}) => { const { data: subscriptionData, error: subscriptionError } = useSubscription(subscription, options); const { data: queryData, error: queryError } = useQuery(query, { ...options, skip: !!subscriptionData && !subscriptionError, }); const data = subscriptionData || queryData; const error = subscriptionError || queryError; const loading = !subscriptionData && !queryData && !error; return { data, loading, error }; }; // Example usage const PostList = () => { const { data, loading, error, loadMore, hasNextPage } = usePaginatedQuery(GET_POSTS, { variables: { first: 10 }, }); const { mutate: createPost, loading: creating } = useOptimisticMutation(CREATE_POST, { optimisticResponse: (variables) => ({ createPost: { __typename: 'Post', id: 'temp-id', ...variables.input, createdAt: new Date().toISOString(), }, }), }); const { data: realTimeData } = useSubscriptionWithFallback( POSTS_SUBSCRIPTION, GET_POSTS ); if (loading && !data) return <div>Loading...</div>; if (error) return <div>Error: {error.message}</div>; const posts = realTimeData?.posts || data?.posts || []; return ( <div> <h1>Posts</h1> <button onClick={() => createPost({ input: { title: 'New Post', content: 'Content' } })} disabled={creating}> {creating ? 'Creating...' : 'Create Post'} </button> <div> {posts.map((post) => ( <div key={post.id}> <h3>{post.title}</h3> <p>{post.content}</p> </div> ))} </div> {hasNextPage && ( <button onClick={loadMore} disabled={loading}> {loading ? 'Loading...' : 'Load More'} </button> )} </div> ); };
